# Исключения: слова, которые НЕ считать запрещёнными.
# Синтаксис тот же, что в словарях, поэтому одна основа закрывает все формы:
#   слово     - основа: (приставка) + основа + любое окончание
#   ~слово    - слово с падежным окончанием и ничем больше
#   =слово    - только точное совпадение
#   re:...    - своё регулярное выражение
# Исключение сильнее словаря: совпало здесь - слово не глушится.

# --- к основам на «хер» ---
херувим
херес         # херес, хересный, хересом
херсон

# --- к основам на «бля» ---
бляш          # бляшка, бляшки, бляшек
блях          # бляха, бляхи, бляхой

# --- к основам «манд», «муд» ---
мандарин
мандат
мандраж
мандрагор
мудр          # мудрый, мудрость, мудрец, мудреный

# --- к основам «еб», «ебо», «еби» (приставка + основа ловит лишнее) ---
себе          # себе, себестоимость
себя
веб           # веб, вебка, вебинар, вебмастер, вебм
требе
потребе
зебр          # зебра, зебры, зебрам
ребен         # ребенок, ребенка, ребенком
ребят         # ребята, ребятам

# --- к основе «трах» ---
трахе         # трахея, трахеи
страх         # страх, страховка, страхование, страховой
страш         # страшно, страшный

# --- к основам списка площадок ---
педикюр
педиатр
re:^героин[яеию]   # героиня, героине, героиню, героиней - но не наркотик
скотч
класс         # класс, классика, классный
секц
сектор
секрет
наркоз
жидк          # жидкость, жидкий, жидко
re:^негромк      # негромко, негромкий - но «негром» глушим
зарезерв
убира         # убирать, убирает, убирайся
re:^насил[ьу]      # насилу, насильно - но не «насиловать»
метод

class
classic
assist
assess
asset
analy         # analysis, analyst
cocktail
cockpit
dickens
scunthorpe
